<#
.SYNOPSIS
    Finds - and optionally repairs - metadata values corrupted by the legacy
    fixed-offset line wrapping in delivered ELAR INDX files.

.DESCRIPTION
    IndxBuilder.createIndx writes the serialized XML in fixed-length slices counted
    from the start of the file, appending a newline after each slice regardless of
    what sits at that offset. Three outcomes:

      - break inside the Base64 payload of ELAR:Content
            harmless, Base64 decoders ignore whitespace
      - break inside element character data
            the metadata value silently gains a line feed ("PDF" -> "P<LF>DF")
      - break inside a tag or attribute name
            the file is not well-formed XML at all

    Detection and repair both use a single streaming character scanner rather than
    an XML parser, so malformed files are handled too, and a 150 MB INDX costs a few
    MB of memory.

    Repair rules, applied only where the break is destructive:

      - inside a tag, between two name characters   -> break removed
      - inside a tag, next to whitespace or quotes  -> break replaced with a space
      - inside an attribute value                   -> break removed
      - inside a non-whitespace text node           -> break removed
      - inside whitespace-only text (indentation)   -> left untouched
      - inside ELAR:Content                         -> left untouched

    After joining, a line may exceed MaxLineLength. The scanner re-breaks it at the
    next element boundary, which is a safe position, so the output still respects the
    limit without ever splitting a token.

.PARAMETER Path
    One or more files, directories, or wildcard patterns. Wildcards are supported in
    both the directory and the file part, e.g. "G:\ELAR\OUT\*\*.INDX.*".
    Directories are expanded using -Filter.

.PARAMETER Filter
    Wildcard applied when Path names a directory. Default *INDX*.

.PARAMETER Recurse
    Recurse into subdirectories when Path names a directory.

.PARAMETER Fix
    Repair the files in place. Without this switch the script only reports.

.PARAMETER NoBackup
    Skip the .bak copy when repairing. Not recommended.

.PARAMETER MaxLineLength
    Maximum output line length used when re-breaking repaired lines. Default 25000.

.PARAMETER Encoding
    Charset used to read and rewrite. Defaults to windows-1252, which is what the
    legacy writer actually emitted; the XML declaration claims ISO-8859-1 and the two
    differ only over 0x80-0x9F. The declaration is left untouched by this script.

.PARAMETER ContentElement
    Local name of the element holding the Base64 payload. Default Content.

.PARAMETER ShowValues
    Include offending values in the report, with breaks rendered as <LF>. Values may
    contain personal data.

.PARAMETER CsvOut
    Optional CSV report path.

.EXAMPLE
    .\Repair-ElarIndxLineBreaks.ps1 -Path "G:\ELAR\OUT\CMOD\S210967_CLICT"

.EXAMPLE
    .\Repair-ElarIndxLineBreaks.ps1 -Path "G:\ELAR\OUT\*\*.INDX.*" -CsvOut .\findings.csv

.EXAMPLE
    .\Repair-ElarIndxLineBreaks.ps1 -Path "G:\ELAR\OUT\CMOD\*" -Recurse -Fix

.NOTES
    Windows PowerShell 5.1 and PowerShell 7 compatible.
    Without -Fix the script never writes to the scanned files.
    With -Fix it writes a temporary file alongside the original and swaps it in only
    after a clean close, so an interrupted run never leaves a truncated INDX.
#>

[CmdletBinding(SupportsShouldProcess = $true, ConfirmImpact = 'High')]
param(
    [Parameter(Mandatory = $true, Position = 0)]
    [string[]] $Path,

    [string] $Filter = '*INDX*',

    [switch] $Recurse,

    [switch] $Fix,

    [switch] $NoBackup,

    [int] $MaxLineLength = 25000,

    [string] $Encoding = 'windows-1252',

    [string] $ContentElement = 'Content',

    [switch] $ShowValues,

    [string] $CsvOut
)

Set-StrictMode -Version 2.0
$ErrorActionPreference = 'Stop'

# ---------------------------------------------------------------------------
# File resolution: wildcards in any position, directories expanded with -Filter
# ---------------------------------------------------------------------------

function Resolve-IndxFiles {
    param([string[]] $Spec, [string] $Filter, [bool] $Recurse)

    $seen  = New-Object 'System.Collections.Generic.HashSet[string]'
    $files = New-Object 'System.Collections.Generic.List[System.IO.FileInfo]'

    foreach ($s in $Spec) {

        $candidates = @()

        if (Test-Path -LiteralPath $s -PathType Container) {
            $gci = @{ LiteralPath = $s; File = $true; Filter = $Filter }
            if ($Recurse) { $gci['Recurse'] = $true }
            $candidates = Get-ChildItem @gci -ErrorAction SilentlyContinue
        }
        elseif (Test-Path -LiteralPath $s -PathType Leaf) {
            $candidates = Get-Item -LiteralPath $s
        }
        else {
            # Wildcard anywhere in the path.
            $gci = @{ Path = $s; File = $true }
            if ($Recurse) { $gci['Recurse'] = $true }
            $candidates = Get-ChildItem @gci -ErrorAction SilentlyContinue

            # A pattern that resolves to directories: expand each with -Filter.
            $dirs = Get-Item -Path $s -ErrorAction SilentlyContinue |
                    Where-Object { $_.PSIsContainer }
            foreach ($d in $dirs) {
                $g2 = @{ LiteralPath = $d.FullName; File = $true; Filter = $Filter }
                if ($Recurse) { $g2['Recurse'] = $true }
                $candidates += Get-ChildItem @g2 -ErrorAction SilentlyContinue
            }
        }

        foreach ($c in $candidates) {
            if ($c -and -not $c.PSIsContainer -and $seen.Add($c.FullName)) {
                $files.Add($c)
            }
        }
    }

    return $files
}

# ---------------------------------------------------------------------------
# Tag repair: fix breaks inside markup without disturbing attribute separation
# ---------------------------------------------------------------------------

function Repair-Tag {
    param([string] $Tag, [ref] $Repairs)

    if ($Tag.IndexOf("`n") -lt 0 -and $Tag.IndexOf("`r") -lt 0) { return $Tag }

    $sb       = New-Object System.Text.StringBuilder ($Tag.Length)
    $inQuote  = $false
    $quoteCh  = [char]0
    $nameChar = { param($c) [char]::IsLetterOrDigit($c) -or $c -eq '_' -or $c -eq '-' -or $c -eq '.' -or $c -eq ':' }

    for ($i = 0; $i -lt $Tag.Length; $i++) {
        $ch = $Tag[$i]

        if ($inQuote) {
            if ($ch -eq $quoteCh) { $inQuote = $false; [void]$sb.Append($ch); continue }
        }
        elseif ($ch -eq '"' -or $ch -eq "'") {
            $inQuote = $true; $quoteCh = $ch; [void]$sb.Append($ch); continue
        }

        if ($ch -eq "`r" -or $ch -eq "`n") {

            # Consume a CRLF pair as one break.
            if ($ch -eq "`r" -and ($i + 1) -lt $Tag.Length -and $Tag[$i + 1] -eq "`n") { $i++ }

            $Repairs.Value++

            if ($inQuote) {
                # A break inside an attribute value is injected corruption.
                continue
            }

            $prev = if ($sb.Length -gt 0) { $sb[$sb.Length - 1] } else { [char]0 }
            $next = if (($i + 1) -lt $Tag.Length) { $Tag[$i + 1] } else { [char]0 }

            if ((& $nameChar $prev) -and (& $nameChar $next)) {
                # Split name: rejoin.
                continue
            }
            # Otherwise the break was legal whitespace between attributes.
            if ($prev -ne ' ') { [void]$sb.Append(' ') }
            continue
        }

        [void]$sb.Append($ch)
    }

    return $sb.ToString()
}

# ---------------------------------------------------------------------------
# Main scanner: one pass, states OUTSIDE / INTAG, streaming through Content
# ---------------------------------------------------------------------------

function Invoke-IndxScan {
    param(
        [System.IO.FileInfo]      $File,
        [System.Text.Encoding]    $Enc,
        [string]                  $ContentLocalName,
        [bool]                    $DoFix,
        [string]                  $OutPath,
        [int]                     $LineLimit
    )

    $findings = New-Object 'System.Collections.Generic.List[psobject]'
    $reader = $null; $writer = $null

    $tagRepairs   = 0
    $textRepairs  = 0
    $rebreaks     = 0
    $docCount     = 0
    $lineNo       = 1
    $lineLen      = 0
    $curElement   = ''
    $inContent    = $false

    $tagBuf  = New-Object System.Text.StringBuilder 256
    $textBuf = New-Object System.Text.StringBuilder 512
    $inTag   = $false

    # Emit helper: tracks line length, re-breaks only before a start tag.
    $emit = {
        param([string] $s, [bool] $mayBreakBefore)

        if ($DoFix -and $mayBreakBefore -and $lineLen -gt $LineLimit) {
            $writer.Write("`n"); $script:lineLen = 0; $script:rebreaks++
        }
        if ($DoFix) { $writer.Write($s) }

        $nl = $s.LastIndexOfAny([char[]]@("`n", "`r"))
        if ($nl -ge 0) { $script:lineLen = $s.Length - $nl - 1 }
        else           { $script:lineLen += $s.Length }
    }

    try {
        $reader = New-Object System.IO.StreamReader($File.FullName, $Enc, $false, 65536)
        if ($DoFix) {
            $writer = New-Object System.IO.StreamWriter($OutPath, $false, $Enc, 65536)
        }

        $script:lineLen = 0
        $script:rebreaks = 0

        $buf = New-Object char[] 65536
        while (($n = $reader.Read($buf, 0, $buf.Length)) -gt 0) {
            for ($k = 0; $k -lt $n; $k++) {
                $ch = $buf[$k]

                if ($ch -eq "`n") { $lineNo++ }

                if ($inTag) {
                    [void]$tagBuf.Append($ch)
                    if ($ch -eq '>') {
                        $tag = $tagBuf.ToString()
                        $tagBuf.Length = 0
                        $inTag = $false

                        $before = 0
                        $fixed  = Repair-Tag -Tag $tag -Repairs ([ref]$before)
                        if ($before -gt 0) {
                            $tagRepairs += $before
                            $findings.Add([pscustomobject]@{
                                File = $File.Name; Kind = 'MarkupLineBreak'
                                Element = $curElement; Line = $lineNo; Value = $tag
                            })
                        }

                        # Element name from a start tag.
                        if ($fixed.Length -gt 1 -and $fixed[1] -ne '/' -and $fixed[1] -ne '?' -and $fixed[1] -ne '!') {
                            $m = [regex]::Match($fixed, '^<\s*([^\s/>]+)')
                            if ($m.Success) {
                                $curElement = $m.Groups[1].Value
                                $local = $curElement.Substring($curElement.IndexOf(':') + 1)
                                $inContent = ($local -eq $ContentLocalName)
                                if ($local -eq 'Doc' -or $local -eq 'Document') { $docCount++ }
                                if ($fixed.EndsWith('/>')) { $inContent = $false }
                            }
                        }
                        elseif ($fixed.Length -gt 1 -and $fixed[1] -eq '/') {
                            $inContent = $false
                        }

                        & $emit $fixed $true
                    }
                    continue
                }

                if ($ch -eq '<') {
                    # Flush the text node accumulated since the last '>'.
                    if ($textBuf.Length -gt 0) {
                        $text = $textBuf.ToString()
                        $textBuf.Length = 0

                        $hasBreak = ($text.IndexOf("`n") -ge 0 -or $text.IndexOf("`r") -ge 0)
                        $isBlank  = [string]::IsNullOrWhiteSpace($text)

                        if ($hasBreak -and -not $isBlank) {
                            # A break inside a real value: injected corruption.
                            $textRepairs++
                            $findings.Add([pscustomobject]@{
                                File = $File.Name; Kind = 'TextLineBreak'
                                Element = $curElement; Line = $lineNo; Value = $text
                            })
                            $text = $text -replace "`r`n", '' -replace "[`r`n]", ''
                        }
                        & $emit $text $false
                    }

                    $inTag = $true
                    [void]$tagBuf.Append($ch)
                    continue
                }

                if ($inContent) {
                    # Base64 payload: stream straight through, never buffered.
                    & $emit ([string]$ch) $false
                }
                else {
                    [void]$textBuf.Append($ch)
                }
            }
        }

        if ($textBuf.Length -gt 0) { & $emit $textBuf.ToString() $false }
    }
    finally {
        if ($writer) { $writer.Flush(); $writer.Dispose() }
        if ($reader) { $reader.Dispose() }
    }

    return [pscustomobject]@{
        File        = $File
        Findings    = $findings
        TagRepairs  = $tagRepairs
        TextRepairs = $textRepairs
        Rebreaks    = $script:rebreaks
        DocCount    = $docCount
    }
}

# ---------------------------------------------------------------------------

try   { $enc = [System.Text.Encoding]::GetEncoding($Encoding) }
catch { throw "Unknown charset '$Encoding'. Try windows-1252, ISO-8859-1, or utf-8." }

$files = Resolve-IndxFiles -Spec $Path -Filter $Filter -Recurse:$Recurse.IsPresent
if ($files.Count -eq 0) {
    Write-Warning "No files matched: $($Path -join ', ')"
    return
}

$mode = if ($Fix) { 'REPAIR (writes in place)' } else { 'report only' }
Write-Host ""
Write-Host "Files: $($files.Count)   Mode: $mode   Charset: $Encoding   Skipping <*:$ContentElement>" -ForegroundColor Cyan
Write-Host ""

$all = New-Object 'System.Collections.Generic.List[psobject]'
$totalDocs = 0; $filesChanged = 0; $i = 0

foreach ($f in $files) {
    $i++
    Write-Progress -Activity 'Scanning INDX files' -Status $f.Name -PercentComplete (($i / $files.Count) * 100)

    $tmp = if ($Fix) { Join-Path $f.DirectoryName ($f.Name + '.repair.tmp') } else { $null }

    $r = Invoke-IndxScan -File $f -Enc $enc -ContentLocalName $ContentElement `
                         -DoFix:$Fix.IsPresent -OutPath $tmp -LineLimit $MaxLineLength

    $totalDocs += $r.DocCount
    foreach ($x in $r.Findings) { $all.Add($x) }
    $bad = $r.TagRepairs + $r.TextRepairs

    if ($bad -eq 0) {
        Write-Host ("  OK    {0,-50} {1,6} docs" -f $f.Name, $r.DocCount) -ForegroundColor DarkGray
        if ($Fix -and (Test-Path -LiteralPath $tmp)) { Remove-Item -LiteralPath $tmp -Force }
        continue
    }

    $label = if ($r.TagRepairs -gt 0) { 'MALFORMED' } else { 'BAD' }
    $color = if ($r.TagRepairs -gt 0) { 'Red' } else { 'Yellow' }
    Write-Host ("  {0,-9} {1,-50} {2,6} docs, {3} value(s), {4} markup" -f `
                $label, $f.Name, $r.DocCount, $r.TextRepairs, $r.TagRepairs) -ForegroundColor $color

    if (-not $Fix) { continue }

    if ($PSCmdlet.ShouldProcess($f.FullName, "Repair $bad line break(s) in place")) {
        if (-not $NoBackup) {
            Copy-Item -LiteralPath $f.FullName -Destination ($f.FullName + '.bak') -Force
        }
        # Swap in only after a clean close: an interrupted run leaves the original.
        Move-Item -LiteralPath $tmp -Destination $f.FullName -Force
        $filesChanged++
        Write-Host ("          repaired, {0} line(s) re-broken at element boundaries" -f $r.Rebreaks) -ForegroundColor Green
    }
    else {
        if (Test-Path -LiteralPath $tmp) { Remove-Item -LiteralPath $tmp -Force }
    }
}
Write-Progress -Activity 'Scanning INDX files' -Completed

$textN = @($all | Where-Object { $_.Kind -eq 'TextLineBreak' }).Count
$tagN  = @($all | Where-Object { $_.Kind -eq 'MarkupLineBreak' }).Count

Write-Host ""
Write-Host "--- Summary ---------------------------------------------------" -ForegroundColor Cyan
Write-Host ("Files scanned          : {0}" -f $files.Count)
Write-Host ("Documents seen         : {0}" -f $totalDocs)
Write-Host ("Corrupted values       : {0}" -f $textN)
Write-Host ("Breaks inside markup   : {0}" -f $tagN)
if ($Fix) { Write-Host ("Files repaired         : {0}" -f $filesChanged) }
Write-Host ""

if ($all.Count -eq 0) {
    Write-Host "No line breaks found outside the Base64 payload." -ForegroundColor Green
    Write-Host ""
    return
}

Write-Host "Affected elements, by frequency:" -ForegroundColor Yellow
$all | Group-Object Element | Sort-Object Count -Descending |
    ForEach-Object { Write-Host ("  {0,6}  {1}" -f $_.Count, $_.Name) }
Write-Host ""

if ($ShowValues) {
    Write-Host "Values (may contain personal data - do not persist):" -ForegroundColor Magenta
    foreach ($x in $all) {
        $v = $x.Value -replace "`r", '<CR>' -replace "`n", '<LF>'
        if ($v.Length -gt 120) { $v = $v.Substring(0, 120) + '...' }
        Write-Host ("  {0}:{1}  {2} = {3}" -f $x.File, $x.Line, $x.Element, $v)
    }
    Write-Host ""
}
else {
    Write-Host "Re-run with -ShowValues to see the affected values (PII warning)." -ForegroundColor DarkGray
    Write-Host ""
}

if ($CsvOut) {
    $export = if ($ShowValues) { $all } else { $all | Select-Object File, Kind, Element, Line }
    $export | Export-Csv -LiteralPath $CsvOut -NoTypeInformation -Encoding UTF8
    Write-Host "Report written to $CsvOut" -ForegroundColor Cyan
    Write-Host ""
}

if ($tagN -gt 0) {
    Write-Host "Breaks inside markup mean those files were not well-formed XML." -ForegroundColor Red
    Write-Host "Any conforming consumer would have rejected them outright." -ForegroundColor Red
    Write-Host ""
}

if (-not $Fix) {
    Write-Host "Re-run with -Fix to repair in place (.bak kept unless -NoBackup)." -ForegroundColor Cyan
    Write-Host ""
}
else {
    Write-Host "Repaired files differ from what was delivered to the archive." -ForegroundColor Yellow
    Write-Host "Re-delivery is a separate decision for the archiving team." -ForegroundColor Yellow
    Write-Host ""
}
