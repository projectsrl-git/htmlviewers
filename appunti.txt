<#
.SYNOPSIS
    Finds - and optionally repairs - metadata values corrupted by the legacy
    fixed-offset line wrapping in delivered ELAR INDX files.

.DESCRIPTION
    IndxBuilder.createIndx writes the serialized XML in fixed-length slices counted
    from the start of the file, appending a newline after each slice regardless of
    what sits at that offset. Three outcomes:

      - break inside the Base64 payload of ELAR:Content
            harmless, Base64 decoders ignore whitespace
      - break inside element character data
            the metadata value silently gains a line feed ("PDF" -> "P<LF>DF")
      - break inside a tag or an attribute
            the file is not well-formed XML at all

    Why this is fast: corruption can only occur AT a line ending, and an INDX wrapped
    at 20000-25000 characters has a few thousand lines even at 150 MB. The scanner
    therefore reads lines, not characters, and classifies each break with native
    string operations. A character-by-character loop in PowerShell would be a hundred
    million interpreted iterations per file; this is a few thousand.

    Classification of each line break, from the state at the end of the line:

      - last '<' after last '>'          -> break inside markup      (destructive)
      - tail after last '>' non-blank
        and current element is Content   -> Base64 payload            (harmless)
      - tail after last '>' non-blank    -> break inside a value      (destructive)
      - tail blank                       -> indentation               (harmless)

    Repair joins the destructive breaks and leaves the harmless ones, so indentation
    and Base64 wrapping are preserved exactly as delivered.

.PARAMETER Path
    Files, directories, or wildcard patterns. Wildcards are supported in both the
    directory and the file part, e.g. "G:\ELAR\OUT\*\*.INDX.*". Directories are
    expanded with -Filter.

.PARAMETER Filter
    Wildcard applied when Path names a directory. Default *INDX*.

.PARAMETER Recurse
    Recurse into subdirectories when Path names a directory.

.PARAMETER Fix
    Repair the files in place. Without this switch the script only reports.

.PARAMETER NoBackup
    Skip the .bak copy when repairing. Not recommended.

.PARAMETER Encoding
    Charset used to read and rewrite. Defaults to windows-1252, which is what the
    legacy writer actually emitted; the XML declaration claims ISO-8859-1 and the two
    differ only over 0x80-0x9F. The declaration is left untouched.

.PARAMETER ContentElement
    Local name of the element holding the Base64 payload. Default Content.

.PARAMETER ShowValues
    Include offending values in the report. Values may contain personal data.

.PARAMETER CsvOut
    Optional CSV report path.

.EXAMPLE
    .\Repair-ElarIndxLineBreaks.ps1 -Path "G:\ELAR\OUT\*\*.INDX.*"

.EXAMPLE
    .\Repair-ElarIndxLineBreaks.ps1 -Path "G:\ELAR\OUT\CMOD\S210967_CLICT" -Fix -WhatIf

.NOTES
    Windows PowerShell 5.1 and PowerShell 7 compatible.
    Without -Fix nothing is written. With -Fix the repaired file is built alongside
    the original and swapped in only after a clean close, so an interrupted run
    leaves the original intact.
#>

[CmdletBinding(SupportsShouldProcess = $true, ConfirmImpact = 'High')]
param(
    [Parameter(Mandatory = $true, Position = 0)]
    [string[]] $Path,

    [string] $Filter = '*INDX*',

    [switch] $Recurse,

    [switch] $Fix,

    [switch] $NoBackup,

    [string] $Encoding = 'windows-1252',

    [string] $ContentElement = 'Content',

    [switch] $ShowValues,

    [string] $CsvOut
)

Set-StrictMode -Version 2.0
$ErrorActionPreference = 'Stop'

function Resolve-IndxFiles {
    param([string[]] $Spec, [string] $Filter, [bool] $Recurse)

    $seen  = New-Object 'System.Collections.Generic.HashSet[string]'
    $files = New-Object 'System.Collections.Generic.List[System.IO.FileInfo]'

    foreach ($s in $Spec) {
        $cand = @()

        if (Test-Path -LiteralPath $s -PathType Container) {
            $g = @{ LiteralPath = $s; File = $true; Filter = $Filter }
            if ($Recurse) { $g['Recurse'] = $true }
            $cand = Get-ChildItem @g -ErrorAction SilentlyContinue
        }
        elseif (Test-Path -LiteralPath $s -PathType Leaf) {
            $cand = Get-Item -LiteralPath $s
        }
        else {
            $g = @{ Path = $s; File = $true }
            if ($Recurse) { $g['Recurse'] = $true }
            $cand = @(Get-ChildItem @g -ErrorAction SilentlyContinue)

            foreach ($d in @(Get-Item -Path $s -ErrorAction SilentlyContinue | Where-Object { $_.PSIsContainer })) {
                $g2 = @{ LiteralPath = $d.FullName; File = $true; Filter = $Filter }
                if ($Recurse) { $g2['Recurse'] = $true }
                $cand += Get-ChildItem @g2 -ErrorAction SilentlyContinue
            }
        }

        foreach ($c in $cand) {
            if ($c -and -not $c.PSIsContainer -and $seen.Add($c.FullName)) { $files.Add($c) }
        }
    }
    return $files
}

# Name characters, for deciding whether a break inside markup split a token.
function Test-NameChar {
    param([char] $c)
    return ([char]::IsLetterOrDigit($c) -or $c -eq '_' -or $c -eq '-' -or $c -eq '.' -or $c -eq ':')
}

<#
    Determines the parser state at the END of a line, using only native string
    operations. Returns the element name currently open and whether the line ends
    inside a tag or inside non-blank character data.
#>
function Get-LineEndState {
    param(
        [string] $Line,
        [string] $PrevElement,
        [bool]   $PrevInTag
    )

    $lastGt = $Line.LastIndexOf('>')
    $lastLt = $Line.LastIndexOf('<')

    $endsInTag = if ($lastLt -gt $lastGt) { $true }
                 elseif ($lastLt -lt 0 -and $lastGt -lt 0) { $PrevInTag }
                 else { $false }

    # The element in scope is the last start tag on this line, if any.
    $element = $PrevElement
    $i = $Line.Length
    while ($true) {
        $i = $Line.LastIndexOf('<', [Math]::Max($i - 1, 0))
        if ($i -lt 0) { break }
        if ($i + 1 -lt $Line.Length) {
            $c = $Line[$i + 1]
            if ($c -eq '/') {
                # Closing tag: nothing is open in character-data terms.
                $element = ''
                break
            }
            if ($c -ne '?' -and $c -ne '!') {
                $j = $i + 1
                while ($j -lt $Line.Length -and (Test-NameChar $Line[$j])) { $j++ }
                $element = $Line.Substring($i + 1, $j - $i - 1)
                # A self-closing tag leaves nothing open.
                if ($lastGt -gt $i -and $lastGt -gt 0 -and $Line[$lastGt - 1] -eq '/') { $element = '' }
                break
            }
        }
        if ($i -eq 0) { break }
    }

    # Text following the last '>' on the line.
    $tail = if ($endsInTag) { '' }
            elseif ($lastGt -ge 0) { $Line.Substring($lastGt + 1) }
            else { $Line }

    return [pscustomobject]@{
        EndsInTag   = $endsInTag
        Element     = $element
        TailIsBlank = [string]::IsNullOrWhiteSpace($tail)
        Tail        = $tail
    }
}

function Invoke-IndxScan {
    param(
        [System.IO.FileInfo]   $File,
        [System.Text.Encoding] $Enc,
        [string]               $ContentLocalName,
        [bool]                 $DoFix,
        [string]               $OutPath
    )

    $findings = New-Object 'System.Collections.Generic.List[psobject]'
    $reader = $null; $writer = $null
    $markupBreaks = 0; $valueBreaks = 0; $docCount = 0; $lineNo = 0

    try {
        $reader = New-Object System.IO.StreamReader($File.FullName, $Enc, $false, 1048576)
        if ($DoFix) { $writer = New-Object System.IO.StreamWriter($OutPath, $false, $Enc, 1048576) }

        $prevLine    = $null
        $prevState   = $null
        $prevElement = ''
        $prevInTag   = $false

        while ($null -ne ($line = $reader.ReadLine())) {
            $lineNo++

            # Cheap document counter: start tags on this line.
            if ($line.IndexOf('<ELAR:Doc') -ge 0 -or $line.IndexOf(':Document>') -ge 0) {
                $docCount++
            }

            if ($null -ne $prevLine) {
                # Classify the break BETWEEN prevLine and line.
                $local = if ($prevState.Element) {
                             $prevState.Element.Substring($prevState.Element.IndexOf(':') + 1)
                         } else { '' }

                $kind = if ($prevState.EndsInTag)      { 'MarkupLineBreak' }
                        elseif ($prevState.TailIsBlank) { 'Safe' }
                        elseif ($local -eq $ContentLocalName) { 'Safe' }
                        else { 'TextLineBreak' }

                if ($kind -eq 'Safe') {
                    if ($DoFix) { $writer.Write($prevLine); $writer.Write("`n") }
                }
                else {
                    if ($kind -eq 'MarkupLineBreak') { $markupBreaks++ } else { $valueBreaks++ }

                    $val = if ($kind -eq 'TextLineBreak') {
                               $prevState.Tail + '<LF>' + $line.Substring(0, [Math]::Min(40, $line.Length))
                           } else {
                               $prevLine.Substring([Math]::Max(0, $prevLine.Length - 40)) + '<LF>' +
                               $line.Substring(0, [Math]::Min(40, $line.Length))
                           }

                    $findings.Add([pscustomobject]@{
                        File    = $File.Name
                        Kind    = $kind
                        Element = $prevState.Element
                        Line    = ($lineNo - 1)
                        Value   = $val
                    })

                    if ($DoFix) {
                        $writer.Write($prevLine)
                        # Inside markup the break may have been legal whitespace
                        # between attributes; only rejoin when it split a token.
                        if ($kind -eq 'MarkupLineBreak') {
                            $a = if ($prevLine.Length -gt 0) { $prevLine[$prevLine.Length - 1] } else { [char]0 }
                            $b = if ($line.Length -gt 0) { $line[0] } else { [char]0 }
                            if (-not ((Test-NameChar $a) -and (Test-NameChar $b))) { $writer.Write(' ') }
                        }
                        # For a value break, join with nothing.
                    }
                }
            }

            $prevLine  = $line
            $prevState = Get-LineEndState -Line $line -PrevElement $prevElement -PrevInTag $prevInTag
            $prevElement = $prevState.Element
            $prevInTag   = $prevState.EndsInTag
        }

        if ($DoFix -and $null -ne $prevLine) { $writer.Write($prevLine); $writer.Write("`n") }
    }
    finally {
        if ($writer) { $writer.Flush(); $writer.Dispose() }
        if ($reader) { $reader.Dispose() }
    }

    return [pscustomobject]@{
        File         = $File
        Findings     = $findings
        MarkupBreaks = $markupBreaks
        ValueBreaks  = $valueBreaks
        DocCount     = $docCount
        Lines        = $lineNo
    }
}

# ---------------------------------------------------------------------------

try   { $enc = [System.Text.Encoding]::GetEncoding($Encoding) }
catch { throw "Unknown charset '$Encoding'. Try windows-1252, ISO-8859-1, or utf-8." }

$files = Resolve-IndxFiles -Spec $Path -Filter $Filter -Recurse:$Recurse.IsPresent
if ($files.Count -eq 0) { Write-Warning "No files matched: $($Path -join ', ')"; return }

$mode = if ($Fix) { 'REPAIR (writes in place)' } else { 'report only' }
Write-Host ""
Write-Host "Files: $($files.Count)   Mode: $mode   Charset: $Encoding   Payload element: $ContentElement" -ForegroundColor Cyan
Write-Host ""

$all = New-Object 'System.Collections.Generic.List[psobject]'
$totalDocs = 0; $filesChanged = 0; $i = 0
$sw = [System.Diagnostics.Stopwatch]::StartNew()

foreach ($f in $files) {
    $i++
    Write-Progress -Activity 'Scanning INDX files' -Status $f.Name -PercentComplete (($i / $files.Count) * 100)

    $tmp = if ($Fix) { Join-Path $f.DirectoryName ($f.Name + '.repair.tmp') } else { $null }

    $r = Invoke-IndxScan -File $f -Enc $enc -ContentLocalName $ContentElement `
                         -DoFix:$Fix.IsPresent -OutPath $tmp

    $totalDocs += $r.DocCount
    foreach ($x in $r.Findings) { $all.Add($x) }
    $bad = $r.MarkupBreaks + $r.ValueBreaks

    if ($bad -eq 0) {
        Write-Host ("  OK        {0,-48} {1,6} docs, {2,6} lines" -f $f.Name, $r.DocCount, $r.Lines) -ForegroundColor DarkGray
        if ($Fix -and $tmp -and (Test-Path -LiteralPath $tmp)) { Remove-Item -LiteralPath $tmp -Force }
        continue
    }

    $label = if ($r.MarkupBreaks -gt 0) { 'MALFORMED' } else { 'BAD' }
    $color = if ($r.MarkupBreaks -gt 0) { 'Red' } else { 'Yellow' }
    Write-Host ("  {0,-9} {1,-48} {2} value(s), {3} markup" -f $label, $f.Name, $r.ValueBreaks, $r.MarkupBreaks) -ForegroundColor $color

    if (-not $Fix) { continue }

    if ($PSCmdlet.ShouldProcess($f.FullName, "Repair $bad line break(s) in place")) {
        if (-not $NoBackup) { Copy-Item -LiteralPath $f.FullName -Destination ($f.FullName + '.bak') -Force }
        Move-Item -LiteralPath $tmp -Destination $f.FullName -Force
        $filesChanged++
        Write-Host "            repaired" -ForegroundColor Green
    }
    elseif ($tmp -and (Test-Path -LiteralPath $tmp)) {
        Remove-Item -LiteralPath $tmp -Force
    }
}
Write-Progress -Activity 'Scanning INDX files' -Completed
$sw.Stop()

$textN = @($all | Where-Object { $_.Kind -eq 'TextLineBreak' }).Count
$tagN  = @($all | Where-Object { $_.Kind -eq 'MarkupLineBreak' }).Count

Write-Host ""
Write-Host "--- Summary ---------------------------------------------------" -ForegroundColor Cyan
Write-Host ("Files scanned        : {0}" -f $files.Count)
Write-Host ("Documents seen       : {0}" -f $totalDocs)
Write-Host ("Corrupted values     : {0}" -f $textN)
Write-Host ("Breaks inside markup : {0}" -f $tagN)
if ($Fix) { Write-Host ("Files repaired       : {0}" -f $filesChanged) }
Write-Host ("Elapsed              : {0:n1}s" -f $sw.Elapsed.TotalSeconds)
Write-Host ""

if ($all.Count -eq 0) {
    Write-Host "No line breaks found outside the Base64 payload and indentation." -ForegroundColor Green
    Write-Host ""
    return
}

Write-Host "Affected elements, by frequency:" -ForegroundColor Yellow
$all | Group-Object Element | Sort-Object Count -Descending |
    ForEach-Object { Write-Host ("  {0,6}  {1}" -f $_.Count, $_.Name) }
Write-Host ""

if ($ShowValues) {
    Write-Host "Values (may contain personal data - do not persist):" -ForegroundColor Magenta
    foreach ($x in $all) { Write-Host ("  {0}:{1}  {2} = {3}" -f $x.File, $x.Line, $x.Element, $x.Value) }
    Write-Host ""
}
else {
    Write-Host "Re-run with -ShowValues to see the affected values (PII warning)." -ForegroundColor DarkGray
    Write-Host ""
}

if ($CsvOut) {
    $export = if ($ShowValues) { $all } else { $all | Select-Object File, Kind, Element, Line }
    $export | Export-Csv -LiteralPath $CsvOut -NoTypeInformation -Encoding UTF8
    Write-Host "Report written to $CsvOut" -ForegroundColor Cyan
    Write-Host ""
}

if ($tagN -gt 0) {
    Write-Host "Breaks inside markup mean those files were not well-formed XML." -ForegroundColor Red
    Write-Host "Any conforming consumer would have rejected them outright." -ForegroundColor Red
    Write-Host ""
}

if (-not $Fix) {
    Write-Host "Re-run with -Fix to repair in place (.bak kept unless -NoBackup)." -ForegroundColor Cyan
}
else {
    Write-Host "Repaired files now differ from what was delivered to the archive." -ForegroundColor Yellow
    Write-Host "Re-delivery is a separate decision for the archiving team." -ForegroundColor Yellow
}
Write-Host ""
